WINDOWS
-------
git submodule update --init --recursive && mkdir build && clang src/SDL_shadercross.c src/cli.c -Iinclude -I../SDL/include -Iexternal/SPIRV-Cross -DSDL_SHADERCROSS_DXC -Lexternal -lSDL3 -LC:\VulkanSDK\1.4.321.1\Bin -lspirv-cross-c-shared -LC:/VulkanSDK/1.4.321.1/Lib -lvulkan-1 -ldxil -ldxcompiler -o build/shadercross.exe && copy .\external\windows\dxcompiler.dll build && copy .\external\windows\dxil.dll build && copy .\external\windows\SDL3.dll build && copy .\external\windows\spirv-cross-c-shared.dll build

LINUX
-----
git submodule update --init --recursive && mkdir build && clang -c src/SDL_shadercross.c -Iinclude -I../SDL/include -Iexternal/SPIRV-Cross -o build/SDL_shadercross.o && clang++ -std=c++17 -c src/cli.c -Iinclude -I../SDL/include -Iexternal/SPIRV-Cross -o build/cli.o && clang++ build/SDL_shadercross.o build/cli.o external/linux/libSDL3.a external/linux/libspirv-cross-c.a external/linux/libspirv-cross-core.a external/linux/libspirv-cross-glsl.a external/linux/libspirv-cross-msl.a external/linux/libspirv-cross-hlsl.a external/linux/libspirv-cross-cpp.a external/linux/libspirv-cross-reflect.a -lm -o build/shadercross
